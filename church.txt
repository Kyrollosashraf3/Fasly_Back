-- ============================================================================
-- church.txt  --  multi-tenant SaaS rollout (PostgreSQL / pgAdmin Query Tool)
-- ============================================================================
-- One database now hosts many churches. Every data / log row carries a
-- church_id; the first permission question is always "is this row in my
-- church?", then "is its class one of mine?". Only the platform owner
-- (super_admin) has no church and sees everything.
--
-- Run this once on an existing fasly_db, or on a fresh one after the app has
-- created its tables. The script is idempotent: every step is guarded, so it
-- is safe to re-run. It mirrors app/db/database.py:create_tables() exactly.
--
-- NOTE: run with autocommit ON (pgAdmin: Query Tool -> "Auto-commit"), or run
-- the two ALTER TYPE statements in section 0 as a separate F5 first. On
-- PostgreSQL 12+ the new enum labels are visible to the CHECK constraint in
-- section 8 within the same script; older servers need a second run of 8.
-- ============================================================================


-- ---------------------------------------------------------------------------
-- 0. Role enum: add the two roles the app now knows about.
--    general_admin (stage amin) and abona (church amin / أبونا).
-- ---------------------------------------------------------------------------
ALTER TYPE role_enum ADD VALUE IF NOT EXISTS 'general_admin';
ALTER TYPE role_enum ADD VALUE IF NOT EXISTS 'abona';


-- ---------------------------------------------------------------------------
-- 1. The tenant anchor table. One row per church.
--    is_active + expiry_date let the platform owner suspend / expire a tenant
--    without deleting any of its data.
-- ---------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS churches (
    id          SERIAL PRIMARY KEY,
    name        VARCHAR(200) NOT NULL,
    region      VARCHAR(100),
    address     TEXT,
    phone       VARCHAR(50),
    phone_2     VARCHAR(50),
    is_active   BOOLEAN NOT NULL DEFAULT TRUE,
    expiry_date DATE,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS ix_churches_id         ON churches (id);
CREATE INDEX IF NOT EXISTS ix_churches_name       ON churches (name);
CREATE INDEX IF NOT EXISTS ix_churches_region     ON churches (region);
CREATE INDEX IF NOT EXISTS ix_churches_is_active  ON churches (is_active);
CREATE INDEX IF NOT EXISTS ix_churches_expiry_date ON churches (expiry_date);


-- ---------------------------------------------------------------------------
-- 2. Rebuild the stage-amin class table if an old database still keys it on
--    users.user_id. The new one is keyed on the servant (teacher_id) so the
--    group follows him when his phone changes. WARNING: this drops the legacy
--    assignment rows; they are re-created from the UI / seed afterwards.
-- ---------------------------------------------------------------------------
DO $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM information_schema.columns
        WHERE table_schema = current_schema()
          AND table_name = 'general_admin_classes'
          AND column_name = 'user_id'
    ) THEN
        DROP TABLE IF EXISTS general_admin_classes;

        CREATE TABLE general_admin_classes (
            id         SERIAL PRIMARY KEY,
            church_id  INTEGER NOT NULL REFERENCES churches(id) ON DELETE RESTRICT,
            teacher_id INTEGER NOT NULL REFERENCES teachers(id) ON DELETE CASCADE,
            class_id   INTEGER NOT NULL REFERENCES classes(id)   ON DELETE CASCADE,
            created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
            CONSTRAINT uq_general_admin_classes_teacher_id_class_id
                UNIQUE (teacher_id, class_id)
        );
        CREATE INDEX ix_general_admin_classes_church_id  ON general_admin_classes (church_id);
        CREATE INDEX ix_general_admin_classes_teacher_id ON general_admin_classes (teacher_id);
        CREATE INDEX ix_general_admin_classes_class_id   ON general_admin_classes (class_id);
    END IF;
END $$;


-- ---------------------------------------------------------------------------
-- 2b. Trip attendance sheet (كشف حضور الرحلة). One row per (trip, boy):
--     attend_flag = حضر ولا لا يوم الرحلة، teacher_id = الخادم المسئول عنه.
--     There is no class_id: the class comes from trips.class_id, a historical
--     snapshot that never moves. One boy is filed under one servant per trip.
-- ---------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS trip_attendances (
    id          SERIAL PRIMARY KEY,
    church_id   INTEGER NOT NULL REFERENCES churches(id) ON DELETE RESTRICT,
    trip_id     INTEGER NOT NULL REFERENCES trips(id)    ON DELETE CASCADE,
    teacher_id  INTEGER REFERENCES teachers(id)          ON DELETE SET NULL,
    boy_id      INTEGER NOT NULL REFERENCES boys(id)     ON DELETE CASCADE,
    attend_flag BOOLEAN NOT NULL DEFAULT FALSE,
    created_at  TIMESTAMP NOT NULL DEFAULT now(),
    CONSTRAINT uq_trip_attendances_trip_id_boy_id UNIQUE (trip_id, boy_id)
);

CREATE INDEX IF NOT EXISTS ix_trip_attendances_id                ON trip_attendances (id);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_church_id         ON trip_attendances (church_id);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_trip_id           ON trip_attendances (trip_id);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_teacher_id        ON trip_attendances (teacher_id);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_boy_id            ON trip_attendances (boy_id);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_attend_flag       ON trip_attendances (attend_flag);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_created_at        ON trip_attendances (created_at);
CREATE INDEX IF NOT EXISTS ix_trip_attendances_trip_id_teacher_id ON trip_attendances (trip_id, teacher_id);

-- ---------------------------------------------------------------------------
-- 6. Tenant foreign keys + indexes. Guarded by pg_constraint so re-runs skip
--    the ones create_all() already made on a fresh database.
-- ---------------------------------------------------------------------------
DO $$
DECLARE
    t TEXT;
BEGIN
    FOREACH t IN ARRAY ARRAY[
        'users', 'teachers', 'boys', 'classes', 'groups',
        'general_admin_classes', 'attends', 'follow_ups', 'teacher_checks',
        'tasks', 'task_values', 'trips', 'trip_attendances', 'subscriptions',
        'notification_logs', 'notification_inbox', 'push_devices'
    ]
    LOOP
        IF NOT EXISTS (
            SELECT 1 FROM pg_constraint
            WHERE conname = t || '_church_id_fkey'
              AND conrelid = t::regclass
        ) THEN
            EXECUTE format(
                'ALTER TABLE %I ADD CONSTRAINT %I '
                'FOREIGN KEY (church_id) REFERENCES churches(id) ON DELETE RESTRICT',
                t, t || '_church_id_fkey'
            );
        END IF;
        EXECUTE format('CREATE INDEX IF NOT EXISTS %I ON %I (church_id)', 'ix_' || t || '_church_id', t);
    END LOOP;
END $$;

-- ---------------------------------------------------------------------------
-- 8. Notification dedupe key becomes unique *per church* so two churches may
--    each fire their own "followup:YYYY-MM-DD" on the same day. Manual rows
--    keep dedupe_key NULL and remain allowed many times over.
-- ---------------------------------------------------------------------------
ALTER TABLE notification_logs DROP CONSTRAINT IF EXISTS uq_notification_logs_dedupe_key;

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
        WHERE conname = 'uq_notification_logs_church_dedupe_key'
    ) THEN
        ALTER TABLE notification_logs
            ADD CONSTRAINT uq_notification_logs_church_dedupe_key
            UNIQUE (church_id, dedupe_key);
    END IF;
END $$;


-- ---------------------------------------------------------------------------
-- 9. Rebuild the account/owner CHECK constraint for the tenant rules and the
--    new abona role. Mirrors _ROLE_OWNER_CHECK in app/db/database.py and
--    app/models/user.py. Only rebuilds if the stored definition is outdated.
-- ---------------------------------------------------------------------------
DO $$
DECLARE
    stored TEXT;
BEGIN
    SELECT pg_get_constraintdef(oid) INTO stored
      FROM pg_constraint
     WHERE conname = 'ck_users_role_owner';

    IF stored IS NULL
       OR stored NOT LIKE '%abona%'
       OR stored NOT LIKE '%church_id IS NOT NULL%' THEN

        ALTER TABLE users DROP CONSTRAINT IF EXISTS ck_users_role_owner;
        ALTER TABLE users ADD CONSTRAINT ck_users_role_owner CHECK (
            (role IN ('teacher', 'admin') AND teacher_id IS NOT NULL
             AND boy_id IS NULL AND church_id IS NOT NULL)
            OR (role = 'user' AND boy_id IS NOT NULL AND teacher_id IS NULL
                AND church_id IS NOT NULL)
            OR (role = 'general_admin' AND teacher_id IS NOT NULL
                AND boy_id IS NULL AND church_id IS NOT NULL)
            OR (role = 'abona' AND boy_id IS NULL AND church_id IS NOT NULL)
            OR (role = 'super_admin' AND boy_id IS NULL)
        );
    END IF;
END $$;

-- ---------------------------------------------------------------------------
-- 11. Verify - every tenant table should report NOT NULL (except users) and
--     no rows should be left without a church.
-- ---------------------------------------------------------------------------
SELECT table_name, column_name, is_nullable
  FROM information_schema.columns
 WHERE table_schema = current_schema()
   AND column_name = 'church_id'
 ORDER BY table_name;

SELECT 'users_without_church' AS check_name, COUNT(*) AS rows
  FROM users WHERE church_id IS NULL AND role <> 'super_admin'
UNION ALL
SELECT 'classes_without_church', COUNT(*) FROM classes WHERE church_id IS NULL
UNION ALL
SELECT 'boys_without_church',    COUNT(*) FROM boys    WHERE church_id IS NULL;
SELECT 'trip_attendance_without_church', COUNT(*) FROM trip_attendances WHERE church_id IS NULL;
-- All counts should be 0.

-- Trip attendance sanity: one row per (trip, boy) and church matches the trip.
SELECT 'duplicate_trip_boy_rows' AS check_name, COUNT(*) AS rows
  FROM (
    SELECT trip_id, boy_id FROM trip_attendances
     GROUP BY trip_id, boy_id HAVING COUNT(*) > 1
  ) d;
SELECT 'trip_attendance_wrong_church' AS check_name, COUNT(*) AS rows
  FROM trip_attendances a JOIN trips t ON t.id = a.trip_id
 WHERE a.church_id <> t.church_id;
-- Both counts should be 0.
